faster-whisper>=1.2

# faster-whisper (<=1.2.1) calls av.open(..., metadata_errors="ignore"), a
# kwarg PyAV 19 removed -- with av 19 every transcription dies with
# "TypeError: open() got an unexpected keyword argument 'metadata_errors'".
# faster-whisper only declares av>=11, so cap it here until a faster-whisper
# release stops passing that kwarg.
av>=11,<19

# `onnx_asr.load_model("nemo-parakeet-tdt-0.6b-v3")` does on first use (same
# on-demand-download convenience as faster-whisper's model cache). The `gpu`
# extra (onnxruntime-gpu) is intentionally NOT included here -- it's added by
# requirements-gpu.txt on machines with a detected NVIDIA GPU; without it,
# onnx-asr runs CPUExecutionProvider only (see parakeet_worker.py's provider
# list comment).
onnx-asr[hub]>=0.9
